\begin{frame}{\textbf{1.1. Bài toán Tối ưu hóa Hộp đen (Black-Box Optimization)}}
	\textbf{Thực trạng:}
	Nhiều bài toán thực tế (Y tế, Hàng không, Điều khiển tự động...) quá phức tạp để mô hình hóa trực tiếp.
	
	\textbf{Đặc điểm "Hộp đen":}
	\begin{itemize}
		\item Không có thông tin về cấu trúc hàm mục tiêu $f(x)$.
		\item Không có đạo hàm giải tích.
		\item Chỉ có thể đánh giá $f(x)$ tại các điểm tham số cụ thể.
	\end{itemize}
	
	\begin{block}{Mục tiêu}
		Tìm cực trị toàn cục mà không cần hiểu "bên trong" hàm số hoạt động như thế nào, thường dựa trên các phương pháp heuristic.
	\end{block}
\end{frame}

\begin{frame}{\textbf{1.2. Lược sử các phương pháp tiếp cận}}
	Bài toán này đã được nghiên cứu hơn 50 năm qua với nhiều hướng tiếp cận:
	\begin{itemize}
		\item \textbf{Cổ điển:} Nelder-Mead (Simplex), Quasi-Newton.
		\item \textbf{Lấy cảm hứng sinh học/vật lý:}
		\begin{itemize}
			\item Simulated Annealing (Luyện kim).
			\item Genetic Algorithms (Di truyền học).
			\item Particle Swarm Optimization (Bầy đàn).
		\end{itemize}
		\item \textbf{Evolution Strategies (ES):} 
		Phát triển bởi Rechenberg \& Schwefel (thập niên 60-70). Dùng đột biến (mutation) và chọn lọc (selection) trên các số thực.
	\end{itemize}
\end{frame}

\begin{frame}{\textbf{1.3. Từ Heuristics đến Toán học chặt chẽ}}
	\textbf{CMA-ES (Covariance Matrix Adaptation - ES):}
	\begin{itemize}
		\item Là chuẩn mực vàng của dòng ES.
		\item Sử dụng ma trận hiệp phương sai để mô hình hóa sự phụ thuộc giữa các biến.
		\item \textit{Tuy nhiên:} Dựa nhiều vào heuristics phức tạp, khó phân tích động lực học một cách chính xác.
	\end{itemize}
	
	\vspace{0.3cm}
	\textbf{Yêu cầu:}
	Cần một phương pháp có hiệu năng tương đương CMA-ES nhưng có \textit{nền tảng toán học minh bạch} thay vì các quy tắc tay (hand-crafted rules).
	$\rightarrow$ \textbf{Sự ra đời của Natural Evolution Strategies (NES).}
\end{frame}

\begin{frame}{\textbf{1.4. Ý tưởng cốt lõi của NES}}
	Khác với các phương pháp ES truyền thống (thường chọn lọc cá thể rồi tạo quần thể mới), NES tiếp cận theo hướng \textbf{Gradient Ascent trên Phân phối}:
	
	\begin{columns}
		\begin{column}{0.5\textwidth}
			\textbf{Cơ chế hoạt động:}
			\begin{enumerate}
				\item Duy trì một phân phối tìm kiếm $\pi(z|\theta)$ (VD: Gaussian).
				\item Cập nhật tham số $\theta$ (VD: $\mu, \Sigma$) theo hướng tăng kỳ vọng hàm thích nghi.
				\item Sử dụng \textbf{Natural Gradient} để đảm bảo sự ổn định.
			\end{enumerate}
		\end{column}
		\begin{column}{0.5\textwidth}
			\begin{exampleblock}{Sự khác biệt}
				NES không tối ưu hóa điểm dữ liệu. NES tối ưu hóa \textbf{phân phối} sinh ra dữ liệu đó.
			\end{exampleblock}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{\textbf{1.5. Tiếp}}

		\item Tại sao Gradient thông thường lại thất bại trong việc cập nhật tham số phân phối (đặc biệt là phương sai)?
	\end{enumerate}

% --- 2.1 Định nghĩa Search Gradient ---
\begin{frame}{\textbf{2.1. Định nghĩa Search Gradient}}
	Để giải quyết bài toán hộp đen một cách toán học, NES tối ưu hóa \textbf{Kỳ vọng hàm thích nghi} (Expected Fitness) thay vì tối ưu trực tiếp $f(x)$.
	
	Gọi $\theta$ là tham số của phân phối tìm kiếm $\pi(z|\theta)$. Hàm mục tiêu $J(\theta)$ được định nghĩa (Eq. 1):
	\begin{equation}
		J(\theta) = \E_{\theta}[f(z)] = \int f(z) \pi(z|\theta) dz
	\end{equation}
	
	\textbf{Vấn đề:} Chúng ta cần tìm $\nabla_\theta J(\theta)$ để thực hiện Gradient Ascent:
	\[ \theta \leftarrow \theta + \eta \nabla_\theta J(\theta) \]
	Nhưng $f(z)$ không khả vi hoặc không biết công thức. Làm sao tính đạo hàm?
\end{frame}

% --- 2.2 Log-Likelihood Trick ---
\begin{frame}{\textbf{2.2. Log-Likelihood Trick}}
	Đây là bước quan trọng nhất để chuyển đạo hàm vào trong phân phối xác suất.
	\begin{align}
		\nabla_\theta J(\theta) &= \nabla_\theta \int f(z) \pi(z|\theta) dz \nonumber \\
		&= \int f(z) \nabla_\theta \pi(z|\theta) dz \quad \text{(Đưa đạo hàm vào trong tích phân)} \nonumber \\
		&= \int f(z) \frac{\nabla_\theta \pi(z|\theta)}{\pi(z|\theta)} \pi(z|\theta) dz \quad \text{(Nhân và chia cho } \pi) \nonumber \\
		&= \int f(z) \nabla_\theta \log \pi(z|\theta) \pi(z|\theta) dz \quad \text{(Quy tắc đạo hàm log)} \nonumber \\
		&= \E_{z \sim \pi} \left[ f(z) \nabla_\theta \log \pi(z|\theta) \right]
	\end{align}
	
	\begin{block}{Kết quả}
		Ta chỉ cần tính đạo hàm của mật độ xác suất $\log \pi$ (thứ ta đã biết), không cần đạo hàm của $f(z)$ (thứ ta không biết).
	\end{block}
\end{frame}

% --- 2.3 Ước lượng Monte Carlo ---
\begin{frame}{\textbf{2.3. Ước lượng Monte Carlo}}
	Trong thực tế, ta không tính tích phân mà xấp xỉ bằng $\lambda$ mẫu thử $z_1, \dots, z_\lambda$ được sinh từ $\pi(\cdot|\theta)$:
	
	\begin{equation}
		\nabla_\theta J(\theta) \approx \frac{1}{\lambda} \sum_{k=1}^{\lambda} f(z_k) \nabla_\theta \log \pi(z_k | \theta)
	\end{equation}
	
	Đây chính là nền tảng của thuật toán \textbf{Canonical Search Gradient}.
	\begin{itemize}
		\item $\nabla_\theta \log \pi(z|\theta)$: Gọi là \textit{Score Function} (cho biết hướng thay đổi tham số để tăng xác suất sinh ra mẫu $z$).
		\item $f(z_k)$: Đóng vai trò trọng số.
	\end{itemize}
\end{frame}

% --- 2.4 Áp dụng cho Gaussian ---
\begin{frame}{\textbf{2.4. Áp dụng cho Phân phối Gaussian đa biến}}
	Xét trường hợp phổ biến nhất: Phân phối chuẩn $d$-chiều $\mathcal{N}(\mu, \Sigma)$.
	\begin{itemize}
		\item Tham số $\theta = \langle \mu, \Sigma \rangle$.
		\item Log-density:
		\[ \log \pi(z|\theta) = -\frac{d}{2}\log(2\pi) - \frac{1}{2}\log \det \Sigma - \frac{1}{2}(z-\mu)^\top \Sigma^{-1} (z-\mu) \]
	\end{itemize}
	
	Đạo hàm riêng (Search Gradients) cụ thể như sau:
	\begin{align}
		\nabla_\mu \log \pi(z) &= \Sigma^{-1}(z - \mu) \\
		\nabla_\Sigma \log \pi(z) &= \frac{1}{2}\Sigma^{-1}(z-\mu)(z-\mu)^\top \Sigma^{-1} - \frac{1}{2}\Sigma^{-1}
	\end{align}
\end{frame}

% --- 2.5 Hạn chế của Plain Gradient ---
\begin{frame}{\textbf{2.5. Tại sao Search Gradient thông thường thất bại?}}
\end{frame}

\begin{frame}{\textbf{2.5.1. Xuất phát điểm: Log-Likelihood}}
	Xét phân phối chuẩn 1 chiều với $\theta = \langle \mu, \sigma \rangle$. Hàm mật độ xác suất (PDF):
	\[ \pi(z|\mu, \sigma) = \frac{1}{\sqrt{2\pi}\sigma} \exp\left( - \frac{(z - \mu)^2}{2\sigma^2} \right) \]
	
	Lấy Logarit tự nhiên ($\log$) để đơn giản hóa việc tính đạo hàm:
	\begin{align*}
		\log \pi(z|\theta) &= \log\left( (2\pi)^{-1/2} \right) + \log(\sigma^{-1}) + \log\left( e^{ - \frac{(z - \mu)^2}{2\sigma^2} } \right) \\
		&= \underbrace{-\frac{1}{2}\log(2\pi)}_{\text{Hằng số } C} - \log(\sigma) - \frac{1}{2}\sigma^{-2}(z - \mu)^2
	\end{align*}
	
	\textbf{Hàm mục tiêu cần đạo hàm:}
	\[ L = C - \log(\sigma) - \frac{1}{2\sigma^2}(z - \mu)^2 \]
\end{frame}

\begin{frame}{\textbf{2.5.2. Biến đổi 1: Đạo hàm theo $\mu$}}
	Ta lấy đạo hàm riêng của $L$ theo $\mu$. Áp dụng quy tắc chuỗi (chain rule) cho cụm $(z-\mu)^2$:
	\[ \frac{\partial L}{\partial \mu} = \frac{\partial}{\partial \mu} \left[ - \frac{1}{2\sigma^2}(z - \mu)^2 \right] \]
	
	Đặt $u = z - \mu \Rightarrow \frac{du}{d\mu} = -1$. Ta có:
	\begin{align*}
		\frac{\partial L}{\partial \mu} &= -\frac{1}{2\sigma^2} \cdot \frac{d}{du}(u^2) \cdot \frac{du}{d\mu} \\
		&= -\frac{1}{2\sigma^2} \cdot (2u) \cdot (-1) \\
		&= \frac{u}{\sigma^2} = \frac{z - \mu}{\sigma^2}
	\end{align*}
	
	\textbf{Kết quả 1:} $\nabla_\mu \log \pi(z) = \frac{z - \mu}{\sigma^2}$
\end{frame}

\begin{frame}{\textbf{2.5.3. Biến đổi 2: Đạo hàm theo $\sigma$}}
	Hàm mục tiêu: $L = C - \log(\sigma) - \frac{1}{2}(z - \mu)^2 \sigma^{-2}$.
	
	Đạo hàm từng thành phần theo $\sigma$:
	\begin{itemize}
		\item Đạo hàm của $-\log(\sigma)$ là: $-\frac{1}{\sigma}$
		\item Đạo hàm của $\sigma^{-2}$ là: $-2\sigma^{-3}$
	\end{itemize}
	
	Thay vào biểu thức:
	\begin{align*}
		\frac{\partial L}{\partial \sigma} &= -\frac{1}{\sigma} - \frac{1}{2}(z - \mu)^2 \cdot (-2\sigma^{-3}) \\
		&= -\frac{1}{\sigma} + \frac{(z - \mu)^2}{\sigma^3}
	\end{align*}
	
	Quy đồng mẫu số chung $\sigma^3$:
	\[ \nabla_\sigma \log \pi(z) = \frac{-\sigma^2 + (z - \mu)^2}{\sigma^3} = \frac{(z - \mu)^2 - \sigma^2}{\sigma^3} \]
\end{frame}

\begin{frame}{\textbf{2.5.4. Tại sao Search Gradient thông thường thất bại?}}
	Kết quả Gradient đối với $\sigma$:
	\[ \nabla_\sigma J \propto \frac{(z - \mu)^2 - \sigma^2}{\sigma^3} \]
	
	\textbf{Sự mất ổn định:}
	\begin{enumerate}
		\item \textbf{Yêu cầu hội tụ:} Khi thuật toán tinh chỉnh lời giải, $\sigma$ cần giảm dần về 0 ($\sigma \to 0$).
		\item \textbf{Nghịch lý toán học:} Mẫu số là $\sigma^3$. Khi $\sigma$ tiến về 0, giá trị $\frac{1}{\sigma^3}$ tiến ra vô cùng ($\infty$).
		\item \textbf{Hậu quả thực tế:} Gradient trở nên "bùng nổ". Bước cập nhật $\theta_{new} = \theta - \eta \nabla$ quá lớn khiến thuật toán văng xa khỏi cực trị.
	\end{enumerate}
\end{frame}

% --- 2.6 Natural Gradient (Giải pháp) ---
\begin{frame}{\textbf{2.6. Giải pháp: Natural Gradient}}
	Thay vì dùng khoảng cách Euclidean trong không gian tham số, ta dùng khoảng cách giữa các phân phối xác suất (Kullback-Leibler Divergence).
	
	\textbf{Bài toán tối ưu ràng buộc:}
	Tìm bước nhảy $\delta \theta$ sao cho tối đa hóa hàm mục tiêu, nhưng giới hạn sự thay đổi của phân phối:
	\begin{align}
		\max_{\delta \theta} \quad & J(\theta + \delta \theta) \approx J(\theta) + \delta \theta^\top \nabla_\theta J \nonumber \\
		\text{s.t.} \quad & D_{KL}(\pi(\cdot|\theta + \delta \theta) || \pi(\cdot|\theta)) = \varepsilon
	\end{align}
	
	Trong đó $D_{KL} \approx \frac{1}{2} \delta \theta^\top F(\theta) \delta \theta$ (xấp xỉ Taylor bậc 2).
\end{frame}

% --- 2.7 Ma trận Fisher ---
\begin{frame}{\textbf{2.7. Ma trận Thông tin Fisher (F)}}
	Giải bài toán tối ưu trên bằng nhân tử Lagrange, ta thu được hướng cập nhật tối ưu (Natural Gradient):
	\begin{equation}
		\tilde{\nabla}_\theta J = F^{-1} \nabla_\theta J
	\end{equation}
	
	Với $F$ là ma trận Fisher, được định nghĩa là hiệp phương sai của score function:
	\begin{equation}
		F = \E \left[ \nabla_\theta \log \pi(z|\theta) \nabla_\theta \log \pi(z|\theta)^\top \right]
	\end{equation}
	
	\textbf{Ý nghĩa của $F^{-1}$:}
	\begin{itemize}
		\item Nó đóng vai trò như một bộ "tiền điều kiện" (Pre-conditioner).
		\item Nó triệt tiêu ảnh hưởng của độ cong không gian tham số (ví dụ: nhân tử $1/\sigma^3$ sẽ bị $F$ triệt tiêu), giúp bước cập nhật ổn định bất kể $\sigma$ lớn hay nhỏ.
	\end{itemize}
\end{frame}

% --- 2.8 Algorithm 3 (Canonical NES) ---
\begin{frame}{\textbf{2.8. Tổng hợp: Thuật toán Canonical NES}}
	\begin{algorithm}[H]
		\begin{algorithmic}[1]
			\STATE \textbf{Input:} $f, \theta_{init}$, learning rate $\eta$
			\REPEAT
			\FOR{$k=1 \dots \lambda$}
			\STATE Draw $z_k \sim \pi(\cdot|\theta)$
			\STATE Evaluate $f(z_k)$
			\STATE Calculate gradients $\nabla_\theta \log \pi(z_k|\theta)$
			\ENDFOR
			\STATE $\nabla_\theta J \leftarrow \frac{1}{\lambda} \sum_{k=1}^\lambda f(z_k) \nabla_\theta \log \pi(z_k|\theta)$
			\STATE $F \leftarrow \frac{1}{\lambda} \sum_{k=1}^\lambda \nabla_\theta \log \pi(z_k|\theta) \nabla_\theta \log \pi(z_k|\theta)^\top$
			\STATE $\theta \leftarrow \theta + \eta F^{-1} \nabla_\theta J$ \COMMENT{Natural Update Step}
			\UNTIL{stopping criterion met}
		\end{algorithmic}
		\caption{Canonical Natural Evolution Strategies}
	\end{algorithm}
\end{frame}

	\begin{frame}

	\begin{frame}{\textbf{3.1. xNES (Exponential NES)}}
		\textbf{Đặc trưng:} Sử dụng ma trận hiệp phương sai đầy đủ (\textit{Full Covariance}), cho phép học mối tương quan giữa các chiều.
		\vspace{0.2cm}
		
		\textbf{Quy trình chi tiết (xNES Pipeline):}
		\begin{enumerate}
			\item \textbf{Khởi tạo:} Tâm $\mu$ và ma trận hiệp phương sai thông qua $\sigma$ và $B$ (với $A = \sigma B$ và $B$ có $\det(B)=1$).
			\item \textbf{Sinh mẫu chuẩn:} Lấy $\lambda$ mẫu $s_k \sim \mathcal{N}(0, I)$.
			\item \textbf{Biến đổi mẫu:} Tạo các cá thể $z_k = \mu + \sigma B s_k$.
			\item \textbf{Đánh giá:} Tính $f(z_k)$ và quy đổi sang Utilities $u_k$ dựa trên hạng.
			\item \textbf{Tính Gradient cục bộ:} Tính đạo hàm cho $\mu, \sigma$ và ma trận $B$ trong hệ tọa độ tự nhiên (nơi $F=I$).
			\item \textbf{Cập nhật kiểu nhân (Exponential Update):} 
			\begin{itemize}
				\item $\mu \leftarrow \mu + \eta_\mu \sigma B \nabla_\delta$
				\item $\sigma \leftarrow \sigma \cdot \exp(\frac{\eta_\sigma}{2} \nabla_\sigma)$
				\item $B \leftarrow B \cdot \exp(\eta_B \nabla_B)$
			\end{itemize}
		\end{enumerate}
	\end{frame}
	
	\begin{frame}{\textbf{3.2. Thuật toán xNES (Pseudo-code)}}
		\begin{algorithm}[H]
			\begin{algorithmic}[1]
				\STATE Sample $s_k \sim \mathcal{N}(0, I)$ for $k=1 \dots \lambda$
				\STATE $z_k \leftarrow \mu + \sigma B s_k$
				\STATE Compute gradients for local coordinates:
				\STATE $\nabla_\delta \leftarrow \sum u_k s_k$, \quad $\nabla_\sigma \leftarrow \sum u_k (\|s_k\|^2 - d)/d$
				\STATE $\nabla_B \leftarrow \sum u_k (s_k s_k^T - I - \frac{\|s_k\|^2 - d}{d} I)$
				\STATE $\mu \leftarrow \mu + \eta_\mu \sigma B \nabla_\delta$
				\STATE $\sigma \leftarrow \sigma \cdot \exp(\frac{\eta_\sigma}{2} \nabla_\sigma)$
				\STATE $B \leftarrow B \cdot \exp(\eta_B \nabla_B)$ \COMMENT{Matrix exponential}
			\end{algorithmic}
			\caption{Exponential NES (xNES)}
		\end{algorithm}
	\end{frame}
	
	\begin{frame}{\textbf{3.3. SNES (Separable NES)}}
		\textbf{Đặc trưng:} Giả định các biến độc lập (\textit{Diagonal Covariance}). Phù hợp cho bài toán rất lớn (high-dimensional).
		\vspace{0.2cm}
		\textbf{Quy trình chi tiết (SNES Pipeline):}
		\begin{enumerate}
			\item \textbf{Khởi tạo:} Vector trung bình $\mu$ và vector độ lệch chuẩn $\sigma$ (cùng số chiều $d$).
			\item \textbf{Sinh mẫu:} Lấy $\lambda$ mẫu $s_k \sim \mathcal{N}(0, I)$.
			\item \textbf{Biến đổi mẫu:} $z_{k,i} = \mu_i + \sigma_i s_{k,i}$ (tính toán độc lập trên từng chiều $i$).
			\item \textbf{Đánh giá:} Tính $f(z_k)$ và quy đổi sang Utilities $u_k$.
			\item \textbf{Tính Gradient thành phần:} 
			\begin{itemize}
				\item $\nabla_{\mu,i} = \sum u_k s_{k,i}$
				\item $\nabla_{\sigma,i} = \sum u_k (s_{k,i}^2 - 1)$
			\end{itemize}
			\item \textbf{Cập nhật độc lập:}
			\begin{itemize}
				\item $\mu_i \leftarrow \mu_i + \eta_\mu \sigma_i \nabla_{\mu,i}$
				\item $\sigma_i \leftarrow \sigma_i \cdot \exp(\frac{\eta_\sigma}{2} \nabla_{\sigma,i})$
			\end{itemize}
		\end{enumerate}
	\end{frame}
	
	\begin{frame}{\textbf{3.4. Thuật toán SNES (Pseudo-code)}}
		\begin{algorithm}[H]
			\begin{algorithmic}[1]
				\STATE Sample $s_k \sim \mathcal{N}(0, I)$ for $k=1 \dots \lambda$
				\STATE $z_{k,i} \leftarrow \mu_i + \sigma_i s_{k,i}$ for each dimension $i$
				\STATE Compute element-wise gradients:
				\STATE $\nabla_{\mu,i} \leftarrow \sum_{k=1}^\lambda u_k s_{k,i}$
				\STATE $\nabla_{\sigma,i} \leftarrow \sum_{k=1}^\lambda u_k (s_{k,i}^2 - 1)$
				\STATE $\mu_i \leftarrow \mu_i + \eta_\mu \sigma_i \nabla_{\mu,i}$
				\STATE $\sigma_i \leftarrow \sigma_i \cdot \exp(\frac{\eta_\sigma}{2} \nabla_{\sigma,i})$
			\end{algorithmic}
			\caption{Separable NES (SNES)}
		\end{algorithm}
	\end{frame}
	
	\begin{frame}{\textbf{3.5. So sánh xNES vs SNES}}
		\centering
		\begin{table}
			\begin{tabular}{l p{3.5cm} p{3.5cm}}
				\toprule
				\textbf{Đặc điểm} & \textbf{xNES (Exponential)} & \textbf{SNES (Separable)} \\
				\midrule
				\textbf{Cấu trúc $\Sigma$} & Ma trận đầy đủ & Ma trận đường chéo \\
				\textbf{Độ phức tạp} & $O(d^3)$ & $O(d)$ \\
				\textbf{Học tương quan} & Có (Rất tốt) & Không \\
				\textbf{Khả năng mở rộng} & Bài toán vừa ($d < 500$) & Bài toán lớn ($d > 10^6$) \\
				\textbf{Tính bất biến} & Bất biến với phép quay & Không bất biến với phép quay \\
				\bottomrule
			\end{tabular}
		\end{table}
	\end{frame}

	\begin{frame}{\textbf{4.1. NES vs CMA-ES}}
		Cả hai đều là tiêu chuẩn vàng trong tối ưu hóa tiến hóa Gaussian.
		\begin{itemize}
			\item \textbf{Giống nhau:} Dùng phân phối Gaussian, dùng rank-based weights.
			\item \textbf{Khác nhau:}
			\begin{itemize}
				\item \textbf{CMA-ES:} Dựa trên Heuristics (Evolution paths). Cực kỳ mạnh trong thực tế nhưng thiếu nền tảng toán học thống nhất.
				\item \textbf{NES:} Dựa trên Natural Gradient (Fisher). Có cơ sở lý thuyết vững chắc, dễ mở rộng sang các phân phối khác.
			\end{itemize}
		\end{itemize}
		\begin{exampleblock}{Kết luận}
			CMA-ES là lựa chọn thực dụng; NES là lựa chọn cho nghiên cứu sâu và các bài toán cần tính bất biến tuyệt đối.
		\end{exampleblock}
	\end{frame}
	
	\begin{frame}{\textbf{4.2. Thiết lập thực nghiệm}}
		\textbf{Bài toán:} Tối ưu hóa hàm \textbf{Rastrigin} với $d=2$.
		\[ f(\mathbf{x}) = 10d + \sum_{i=1}^{d} [x_i^2 - 10\cos(2\pi x_i)] \]
		\begin{itemize}
			\item \textbf{Đặc điểm:} Non-convex, cực kỳ nhiều bẫy local minima.
			\item \textbf{Mục tiêu:} Tìm Global Minimum tại $\mathbf{x} = \mathbf{0}, f(\mathbf{x}) = 0$.
		\end{itemize}
		\textbf{Cấu hình thuật toán:}
		\begin{itemize}
			\item \textbf{NES:} Gaussian $\mathcal{N}(\mu, \sigma^2 I)$, $\lambda=50$, Rank-based Utilities.
			\item \textbf{CMA-ES:} Cấu hình chuẩn (thư viện \texttt{pycma}).
			\item \textbf{Gradient Descent (GD):} Learning rate tối ưu, dùng analytic gradient.
		\end{itemize}
	\end{frame}

	\begin{frame}{\textbf{4.3. Kết luận}}
		\begin{enumerate}
			\item NES cung cấp nền tảng toán học vững chắc cho tiến hóa thông qua Hình học thông tin.
			\item Natural Gradient là chìa khóa để đạt được sự ổn định và tính bất biến.
			\item xNES mang lại độ chính xác cao nhất, SNES mang lại hiệu suất cho các bài toán lớn.
