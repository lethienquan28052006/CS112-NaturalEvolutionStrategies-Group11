	\begin{frame}{\textbf{3.3. Hệ tọa độ tự nhiên (Natural Coordinates)}}
		Định nghĩa phép cập nhật tham số thông qua ánh xạ mũ (Exponential Map):
		
		\begin{enumerate}
			\item \textbf{Cập nhật Vị trí:} $\mu_{\text{new}} = \mu + A \cdot \delta$ \quad (9)
			\item \textbf{Cập nhật Ma trận:} $A_{\text{new}} = A \cdot \exp\left(\frac{1}{2}M\right)$ \quad (10)
		\end{enumerate}
		
		\begin{explanation}
			\textbf{Tại sao dùng hàm mũ $\exp(M)$?}
			\begin{itemize}
				\item Ma trận hiệp phương sai $\Sigma = A A^T$ bắt buộc phải là \textit{đối xứng xác định dương}.
				\item Nếu cộng trực tiếp ma trận ($A + \Delta A$), ta không đảm bảo được tính chất này.
				\item Hàm mũ ma trận $\exp(X)$ luôn trả về ma trận xác định dương với mọi $X$ đối xứng. Do đó, công thức (10) đảm bảo $A_{new}$ luôn hợp lệ về mặt toán học mà không cần điều kiện ràng buộc nào khác.
				\item Hệ số $\frac{1}{2}$ xuất hiện để khớp với định nghĩa $\Sigma_{new} = A_{new} A_{new}^T = A e^{M/2} e^{M/2} A^T = A e^M A^T$.
			\end{itemize}
		\end{explanation}
	\end{frame}
	
	% --- SLIDE 3.4 ---
	\begin{frame}{\textbf{3.4. Log-Likelihood trong tọa độ tự nhiên}}
		Thay thế (9), (10) vào log-likelihood của phân phối chuẩn và dùng Taylor bậc 1:
		
		\[ J(\delta, M) \approx C - \underbrace{\frac{1}{2}\text{tr}(M)}_{\text{Phần 1}} - \underbrace{\frac{1}{2} \| (I - \frac{1}{2}M)(z - \delta) \|^2}_{\text{Phần 2}} \]
		
		\begin{explanation}
			\textbf{Các bước biến đổi cụ thể:}
			\begin{enumerate}
				\item \textbf{Phần 1 (Từ định thức):} $\log(\det(A_{new})) = \text{tr}(\log A_{new})$.
				\\ Thay (10): $\text{tr}(\log(A \cdot e^{M/2})) = \text{tr}(\log A + \frac{1}{2}M) = \text{Const} + \frac{1}{2}\text{tr}(M)$.
				\item \textbf{Phần 2 (Từ khoảng cách):} $x - \mu_{new} = (\mu + Az) - (\mu + A\delta) = A(z-\delta)$.
				\\ Ma trận nghịch đảo: $A_{new}^{-1} = (A e^{M/2})^{-1} = e^{-M/2} A^{-1}$.
				\\ Khoảng cách chuẩn hóa: $A_{new}^{-1}(x-\mu_{new}) = e^{-M/2} A^{-1} A (z-\delta) = e^{-M/2}(z-\delta)$.
				\item \textbf{Xấp xỉ Taylor:} Với $M \approx 0$, ta có $e^{-M/2} \approx I - \frac{1}{2}M$.
				\\ Thay vào bình phương độ dài vector: $\| (I - \frac{1}{2}M)(z-\delta) \|^2$.
			\end{enumerate}
		\end{explanation}
	\end{frame}
	
	% --- SLIDE 3.5 ---
	\begin{frame}{\textbf{3.5. Dẫn xuất Gradient vị trí ($G_\delta$)}}
		Đạo hàm theo $\delta$ tại điểm gốc ($\delta=0, M=0$):
		\[ \nabla_\delta J \approx \nabla_\delta \left( -\frac{1}{2} \|z - \delta\|^2 \right) \Big|_{\delta=0} = z \]
		\vspace{-0.5em}
		\begin{beamercolorbox}[sep=4pt,center,shadow=true,rounded=true, wd=0.5\textwidth]{block body}
			$G_\delta = \sum u_i z_i$
		\end{beamercolorbox}
		
		\begin{explanation}
			\textbf{Chi tiết đạo hàm vector:}
			\begin{itemize}
				\item Đặt $f(\delta) = -\frac{1}{2} \|z - \delta\|^2 = -\frac{1}{2} (z - \delta)^T (z - \delta)$.
				\item Khai triển: $-\frac{1}{2} (z^T z - 2z^T \delta + \delta^T \delta)$.
				\item Đạo hàm theo $\delta$:
				\[ \frac{\partial}{\partial \delta} \dots = -\frac{1}{2} (0 - 2z + 2\delta) = z - \delta \]
				\item Tại điểm bắt đầu cập nhật, ta chưa di chuyển nên $\delta = 0$.
				\[ \Rightarrow \text{Gradient} = z - 0 = z \]
			\end{itemize}
		\end{explanation}
	\end{frame}
	
	% --- SLIDE 3.6 ---
	\begin{frame}{\textbf{3.6. Dẫn xuất Gradient ma trận ($G_M$)}}
		Đạo hàm theo $M$ tại điểm gốc:
		\[ \nabla_M J \approx \nabla_M \left( -\frac{1}{2}\text{tr}(M) - \frac{1}{2} z^T (I - M) z \right) = \frac{1}{2} (z z^T - I) \]
		\vspace{-0.5em}
		\begin{beamercolorbox}[sep=4pt,center,shadow=true,rounded=true, wd=0.5\textwidth]{block body}
			$G_M = \sum u_i (z_i z_i^T - I)$
		\end{beamercolorbox}
		
		\begin{explanation}
			\textbf{Chi tiết đạo hàm ma trận:} (Sử dụng các identity đạo hàm ma trận cơ bản)
			\begin{enumerate}
				\item \textbf{Đạo hàm Trace:} $\frac{\partial}{\partial M} (-\frac{1}{2} \text{tr}(M)) = -\frac{1}{2} I$. (Vì đạo hàm trace(M) là I).
				\item \textbf{Đạo hàm số mũ (đã xấp xỉ):} Ta xét phần chứa $M$ trong biểu thức $\|(I - \frac{1}{2}M)z\|^2 \approx z^T (I-M) z$ (bỏ qua bậc 2 của M).
				\\ Phần phụ thuộc M là: $-\frac{1}{2} z^T (-M) z = \frac{1}{2} z^T M z = \frac{1}{2} \text{tr}(z z^T M)$.
				\\ Đạo hàm của $\text{tr}(A M)$ theo $M$ chính là $A^T$. Ở đây $A = z z^T$.
				\\ $\Rightarrow \text{Gradient} = +\frac{1}{2} z z^T$.
				\item \textbf{Tổng hợp:} $-\frac{1}{2} I + \frac{1}{2} z z^T = \frac{1}{2} (z z^T - I)$.
			\end{enumerate}
		\end{explanation}
	\end{frame}
	
	% --- SLIDE 3.7 ---
	\begin{frame}{\textbf{3.7. Phân rã tham số (Decomposition)}}
		Tách không gian cập nhật thành hai phần trực giao:
		\begin{columns}
			\column{0.5\textwidth}
			1. \textbf{Scale ($G_\sigma$):} Hướng đẳng hướng
			\[ G_\sigma = \frac{\text{tr}(G_M)}{d} \]
			\column{0.5\textwidth}
			2. \textbf{Shape ($G_B$):} Hướng lệch
			\[ G_B = G_M - G_\sigma \cdot I \]
		\end{columns}
		
		\begin{explanation}
			\textbf{Tại sao công thức này đảm bảo trực giao?}
			\begin{itemize}
				\item Ta phân tích ma trận $G_M$ thành tổng: $G_M = s \cdot I + G_B$.
				\item Lấy Trace hai vế: $\text{tr}(G_M) = s \cdot \text{tr}(I) + \text{tr}(G_B) = s \cdot d + \text{tr}(G_B)$.
				\item Yêu cầu của Shape update là \textbf{giữ nguyên thể tích}, tương đương với định thức không đổi $\rightarrow$ $\text{tr}(G_B)$ phải bằng 0.
				\item Thay $\text{tr}(G_B)=0$ vào phương trình trên: $\text{tr}(G_M) = s \cdot d \Rightarrow s = \text{tr}(G_M)/d$.
				\item $\Rightarrow$ Đây là cách duy nhất để tách phần thay đổi kích thước ($s$) ra khỏi phần thay đổi hình dáng ($G_B$).
			\end{itemize}
		\end{explanation}
	\end{frame}
